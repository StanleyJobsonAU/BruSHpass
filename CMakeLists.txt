cmake_minimum_required(VERSION 3.22)
project(BruSHpass VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BP_STATIC_OPENSSL "Link static OpenSSL" ON)
option(BP_CLIENT_TEMPLATE "Build an agent template that refuses to run without provisioning" OFF)
set(OPENSSL_USE_STATIC_LIBS ${BP_STATIC_OPENSSL})
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(OPENSSL_MSVC_STATIC_RT TRUE)
endif()
find_package(OpenSSL 3.0 REQUIRED)
find_package(Threads REQUIRED)
set(BP_SOURCES native/main.c native/platform.c native/identity.c native/protocol.c native/export.c native/execute.c)
add_executable(brushpass ${BP_SOURCES})
set_target_properties(brushpass PROPERTIES OUTPUT_NAME "BruSHpass")
if(BP_CLIENT_TEMPLATE)
    target_compile_definitions(brushpass PRIVATE BP_CLIENT_TEMPLATE=1)
endif()
target_include_directories(brushpass PRIVATE native)
target_link_libraries(brushpass PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
    target_compile_definitions(brushpass PRIVATE _WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN)
    target_link_libraries(brushpass PRIVATE ws2_32 iphlpapi advapi32 crypt32 bcrypt)
endif()
if(MSVC)
    target_compile_options(brushpass PRIVATE /W4)
else()
    target_compile_options(brushpass PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(BP_TEMPLATE_WINDOWS "" CACHE FILEPATH "Prebuilt Windows x64 client template")
set(BP_TEMPLATE_LINUX "" CACHE FILEPATH "Prebuilt Linux x64 client template")
if(BP_TEMPLATE_WINDOWS AND BP_TEMPLATE_LINUX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(BP_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/embedded_templates.c")
    add_custom_command(OUTPUT "${BP_GENERATED}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_templates.py"
            "${BP_TEMPLATE_WINDOWS}" "${BP_TEMPLATE_LINUX}" "${BP_GENERATED}"
        DEPENDS "${BP_TEMPLATE_WINDOWS}" "${BP_TEMPLATE_LINUX}" tools/embed_templates.py VERBATIM)
    target_sources(brushpass PRIVATE "${BP_GENERATED}")
    target_compile_definitions(brushpass PRIVATE BP_WITH_TEMPLATES=1)
else()
    message(STATUS "Building a client template / development server without embedded exporters")
endif()
enable_testing()
add_test(NAME native_protocol COMMAND brushpass --self-test)
